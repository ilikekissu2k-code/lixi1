    <script>
        (function() {
            "use strict";

            // ---------- 🎋 20 câu chúc Tết ý nghĩa (dành cho Mia đáp lễ) ----------
            const WISHES = [
                "năm mới an khang thịnh vượng, vạn sự như ý",
                "năm mới phát tài phát lộc, tiền vào như nước",
                "năm mới sức khỏe dồi dào, bình an bên gia đình",
                "năm mới công việc hanh thông, thăng quan tiến chức",
                "năm mới tràn đầy niềm vui và tiếng cười",
                "năm mới may mắn ngập trời, lộc vàng đầy nhà",
                "năm mới thành công rực rỡ, vỡ òa hạnh phúc",
                "năm mới an nhiên tự tại, mọi điều tốt lành",
                "năm mới tấn tài tấn lộc, phúc lộc song toàn",
                "năm mới vạn điều may mắn, như ý cát tường",
                "năm mới thịnh vượng giàu sang, nhà nhà hạnh phúc",
                "năm mới năng lượng tràn trề, chinh phục mọi đỉnh cao",
                "năm mới bình an vui vẻ, khỏe như voi",
                "năm mới đong đầy yêu thương, vẹn tròn ước mơ",
                "năm mới phúc khí đầy nhà, con đường rộng mở",
                "năm mới tài lộc dồi dào, làm ăn thuận buồm xuôi gió",
                "năm mới sức khỏe an khang, tinh thần sảng khoái",
                "năm mới mọi sự hanh thông, cầu được ước thấy",
                "năm mới vinh hoa phú quý, phước lộc miên trường",
                "năm mới hạnh phúc ngập tràn, sum vầy bên người thân"
            ];

            // ---------- Phân tích lời chúc để chọn cách xưng hô: "anh" hay "em" ----------
            // Người dùng thường là "anh" hoặc "em" so với Mia.
            // Quy tắc:
            // - Nếu người dùng gọi Mia = "chị", "cô", "bác", "dì", "mợ", "thím", "gì", "anh" (coi Mia là anh) => họ nhỏ tuổi hơn => Mia chúc EM.
            // - Nếu người dùng gọi Mia = "em", "cháu", "con", "chắt" => họ lớn tuổi hơn => Mia chúc ANH.
            // - Mặc định: "anh" (lịch sự, phổ biến)
            function getAddressFromGreeting(greeting) {
                const lower = greeting.toLowerCase();

                // Những từ cho thấy người dùng đang vai dưới (gọi Mia bằng vai trên) => Mia chúc EM
                const youngerIndicators = ["chị", "cô", "bác", "dì", "mợ", "thím", "gì", "dượng", "anh"];
                // Những từ cho thấy người dùng đang vai trên (gọi Mia bằng vai dưới) => Mia chúc ANH
                const olderIndicators = ["em", "cháu", "con", "chắt"];

                // Ưu tiên tìm từ ngay sau "chúc" để chính xác hơn
                const chucIndex = lower.indexOf('chúc');
                let targetArea = lower;
                if (chucIndex !== -1) {
                    targetArea = lower.substring(chucIndex + 4);
                }

                // Kiểm tra từ chỉ vai dưới trước (vì nếu có cả "em" và "chị" thì nên ưu tiên ngữ cảnh? 
                // Thực tế, một câu thường chỉ gọi một đối tượng. Ta kiểm tra lần lượt.
                for (let kw of youngerIndicators) {
                    if (targetArea.includes(kw)) {
                        return "em";   // Mia chúc em
                    }
                }
                for (let kw of olderIndicators) {
                    if (targetArea.includes(kw)) {
                        return "anh";  // Mia chúc anh
                    }
                }

                // Không tìm thấy -> mặc định "anh" (thân thiện, tôn trọng)
                return "anh";
            }

            // ---------- Tạo câu chúc ngẫu nhiên với xưng hô đúng ----------
            function generateReply(addressTerm) {
                const randomIndex = Math.floor(Math.random() * WISHES.length);
                const wish = WISHES[randomIndex];
                return `Mia chúc ${addressTerm} ${wish}! ✨`;
            }

            // DOM elements
            const inputSection = document.getElementById('input-section');
            const envelopeSection = document.getElementById('envelope-section');
            const resultSection = document.getElementById('result-section');
            const submitBtn = document.getElementById('submit-btn');
            const greetingInput = document.getElementById('greeting');
            const envelope = document.getElementById('envelope');
            const replyBox = document.getElementById('reply-message');

            let userGreeting = '';  // lưu lời chúc gốc

            // ---------- GỬI LỜI CHÚC: ẩn input, hiện bao lì xì ----------
            submitBtn.addEventListener('click', function(e) {
                e.preventDefault();
                userGreeting = greetingInput.value.trim();

                if (userGreeting === '') {
                    alert('🌸 Vui lòng nhập lời chúc Tết nhé!');
                    return;
                }

                inputSection.classList.add('hidden');
                envelopeSection.classList.remove('hidden');
                resultSection.classList.add('hidden'); // ẩn kết quả cũ nếu có

                // Reset trạng thái bao (chưa mở)
                envelope.classList.remove('open');
            });

            // ---------- MỞ BAO LÌ XÌ: phân tích xưng hô + chúc ngẫu nhiên + hiển thị QR ----------
            envelope.addEventListener('click', function() {
                if (!userGreeting) {
                    alert('Hãy gửi lời chúc trước khi mở lì xì nhé!');
                    return;
                }

                // Hiệu ứng mở bao
                envelope.classList.add('open');

                // Phân tích lời chúc để lấy cách xưng hô (anh/em)
                const address = getAddressFromGreeting(userGreeting);

                // Lấy câu chúc ngẫu nhiên kèm xưng hô
                const replyText = generateReply(address);
                replyBox.innerText = replyText;

                // Hiện khung kết quả (lời chúc + QR)
                resultSection.classList.remove('hidden');
            });

            // ---------- Nhấn Enter (không Shift) trong textarea cũng gửi ----------
            greetingInput.addEventListener('keypress', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    submitBtn.click();
                }
            });

            // ---------- Ẩn các phần phụ khi load trang (chỉ hiện input) ----------
            envelopeSection.classList.add('hidden');
            resultSection.classList.add('hidden');
            inputSection.classList.remove('hidden');

        })();
    </script>
